# verify_predictions.py
import os
import pandas as pd
import yfinance as yf
from datetime import datetime, timedelta
import joblib

# ----------------------------
# Configuration
# ----------------------------
TICKERS = ["AAPL", "GOOGL", "MSFT", "TSLA", "INFY.NS"]
MODEL_DIR = "models"
LOG_FILE = "predictions_summary.csv"

# Features used during training (exact 10 features)
FEATURE_COLS = [
    "SMA_10",
    "SMA_50",
    "EMA_10",
    "EMA_50",
    "RSI",
    "Open",
    "High",
    "Low",
    "Close",
    "Volume",
]


# ----------------------------
# Compute RSI
# ----------------------------
def compute_rsi(series, period=14):
    delta = series.diff()
    gain = (delta.where(delta > 0, 0)).rolling(period).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(period).mean()
    rs = gain / loss
    return 100 - (100 / (1 + rs))


# ----------------------------
# Make prediction for one ticker
# ----------------------------
def make_prediction(ticker):
    model_path = os.path.join(MODEL_DIR, f"{ticker}_xgboost.pkl")
    if not os.path.exists(model_path):
        print(f"[❌] Failed for {ticker}: Model not found at {model_path}")
        return None

    try:
        model = joblib.load(model_path)
    except Exception as e:
        print(f"[❌] Failed to load model for {ticker}: {e}")
        return None

    # Fetch last 90 days of stock data
    end = datetime.today()
    start = end - timedelta(days=90)
    df = yf.download(ticker, start=start, end=end)
    if df.empty:
        print(f"[❌] No data for {ticker}")
        return None
    df.reset_index(inplace=True)

    # Technical indicators
    df["SMA_10"] = df["Close"].rolling(10).mean()
    df["SMA_50"] = df["Close"].rolling(50).mean()
    df["EMA_10"] = df["Close"].ewm(span=10, adjust=False).mean()
    df["EMA_50"] = df["Close"].ewm(span=50, adjust=False).mean()
    df["RSI"] = compute_rsi(df["Close"], 14)

    df = df.dropna()
    if df.empty:
        print(f"[❌] Not enough data for {ticker}")
        return None

    latest = df.iloc[-1]

    # Prepare features
    try:
        features = latest[FEATURE_COLS].values.reshape(1, -1)
    except Exception as e:
        print(f"[❌] Feature mismatch for {ticker}: {e}")
        return None

    # Predict
    try:
        predicted_close = model.predict(features)[0]
    except Exception as e:
        print(f"[❌] Prediction failed for {ticker}: {e}")
        return None

    # ⚡ Correct scalar date extraction
    latest_date = pd.to_datetime(latest["Date"].item())

    return {
        "date": latest_date.strftime("%Y-%m-%d"),
        "stock": ticker,
        "predicted_close": round(float(predicted_close), 2),
        "actual_close": None,
        "error": None,
        "mape(%)": None,
    }


# ----------------------------
# Main
# ----------------------------
def main():
    # Robust loading of log file
    try:
        if os.path.exists(LOG_FILE) and os.path.getsize(LOG_FILE) > 0:
            log_df = pd.read_csv(LOG_FILE)
        else:
            log_df = pd.DataFrame(
                columns=[
                    "date",
                    "stock",
                    "predicted_close",
                    "actual_close",
                    "error",
                    "mape(%)",
                ]
            )
    except pd.errors.EmptyDataError:
        log_df = pd.DataFrame(
            columns=[
                "date",
                "stock",
                "predicted_close",
                "actual_close",
                "error",
                "mape(%)",
            ]
        )

    new_rows = []
    for ticker in TICKERS:
        row = make_prediction(ticker)
        if row:
            # Avoid duplicates
            if not (
                (log_df["date"] == row["date"]) & (log_df["stock"] == row["stock"])
            ).any():
                new_rows.append(row)

    if new_rows:
        log_df = pd.concat([log_df, pd.DataFrame(new_rows)], ignore_index=True)

    log_df.to_csv(LOG_FILE, index=False)
    print("\n[📁] Predictions saved to", LOG_FILE)
    if new_rows:
        print("[➕] Added predictions for:", ", ".join([r["stock"] for r in new_rows]))


if __name__ == "__main__":
    main()
